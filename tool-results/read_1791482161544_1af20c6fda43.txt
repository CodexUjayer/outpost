     1→// ============================================================
     2→// Procedural audio engine — Off-World Colony STEM Game (Task 3-a)
     3→// 100% synthesized with Web Audio API. No files, no fetch, no assets.
     4→//
     5→// Design:
     6→//  - one shared AudioContext; master chain: masterGain -> compressor -> destination
     7→//  - two shared 2s noise buffers (white + brown) reused by every noise sound
     8→//  - persistent ambience layers (wind / storm / room tone / fan) crossfaded
     9→//    with setTargetAtTime — never hard switches
    10→//  - every public method is safe to call any time (no-ops before init()),
    11→//    cheap when called every frame (state caching + epsilon gates)
    12→//  - one-shots capped (~24 concurrent), auto-cleanup via onended
    13→// ============================================================
    14→
    15→export type AudioPosition = { x: number; y: number; z: number }
    16→export type SurfaceType = 'regolith' | 'metal'
    17→export type Planet = 'moon' | 'mars'
    18→
    19→// ---------- tiny helpers ----------
    20→
    21→const clamp01 = (v: number): number => {
    22→  if (typeof v !== 'number' || !isFinite(v)) return 0
    23→  return v < 0 ? 0 : v > 1 ? 1 : v
    24→}
    25→
    26→const num = (v: number, d: number): number =>
    27→  typeof v === 'number' && isFinite(v) ? v : d
    28→
    29→interface LoopParts {
    30→  sources: AudioScheduledSourceNode[]
    31→  cleanup: AudioNode[]
    32→  params: Record<string, AudioParam>
    33→}
    34→
    35→interface LoopNode extends LoopParts {
    36→  id: string
    37→  ctrl: GainNode // fade / master control gain for the loop
    38→  panner: PannerNode | null
    39→}
    40→

    42→// (generators, pumps, compressors, ventilation, hydraulics — downloaded
    43→// from the internet; see tools/fetch-sounds.sh)

    62→  private brown: AudioBuffer | null = null
    63→  private heartGain: GainNode | null = null
    64→  private volume = 0.9
    65→  private doppler = false
    66→
    67→  // ---- ambience nodes (persistent) ----

    69→  private windBrownG: GainNode | null = null
    70→  private windFilter: BiquadFilterNode | null = null
    71→  private windGain: GainNode | null = null
    72→  private stormGain: GainNode | null = null
    73→  private roomGain: GainNode | null = null
    74→  private fanGain: GainNode | null = null
    75→
    76→  // ---- cached applied targets (keeps per-frame calls cheap) ----
    77→  private apPlanet: Planet | null = null
    78→  private apWind = -1
    79→  private apStorm = -1
    80→  private apRoom = -1
    81→  private apFan = -1
    82→  private apBreath = -1
    83→  private apBreathRate = -1
    84→
    85→  // ---- desired state (survives being set before init) ----
    86→  private env = { inside: false, planet: 'moon' as Planet, windIntensity: 0.3, storm: false }
    87→  private fanLevel = 0.4
    88→  private breathOn = false
    89→  private breathEffort = 0.4
    90→  private heartLevel = 0
    91→  private alarmOn = false
    92→  private roverOn = false
    93→  private roverThrottle = 0
    94→  // real recorded rover audio (NASA/JPL Perseverance drive — Sol 16)
    95→  private roverBuf: AudioBuffer | null = null
    96→  private roverBufTried = false

   104→  // real recorded machinery/industrial one-shots (colony ambience)

   106→  private machTried = false
   107→  private motorOn = false
   108→  private motorThrottle = 0
   109→  private burnOn = false

   127→  // ================================================================
   128→  // Lifecycle

   131→  init(): void {
   132→    if (this.ready) {
   133→      this.resume()
   134→      return
   135→    }
   136→    if (typeof window === 'undefined') return
   137→    try {
   138→      const w = window as unknown as {
   139→        AudioContext?: typeof AudioContext
   140→        webkitAudioContext?: typeof AudioContext
   141→      }
   142→      const Ctor = w.AudioContext ?? w.webkitAudioContext
   143→      if (!Ctor) return
   144→
   145→      const ctx = new Ctor()
   146→      this.ctx = ctx
   147→
   148→      // shared noise buffers (2s each)
   149→      this.white = this.makeNoiseBuffer('white')
   150→      this.brown = this.makeNoiseBuffer('brown')
   151→
   152→      // master chain: gain -> compressor -> destination
   153→      const master = ctx.createGain()
   154→      master.gain.value = clamp01(this.volume)
   155→      const comp = ctx.createDynamicsCompressor()
   156→      comp.threshold.value = -16
   157→      comp.knee.value = 18
   158→      comp.ratio.value = 5
   159→      comp.attack.value = 0.004
   160→      comp.release.value = 0.24
   161→      master.connect(comp)
   162→      comp.connect(ctx.destination)
   163→      this.master = master
   164→
   165→      // heartbeat bus
   166→      const hg = ctx.createGain()
   167→      hg.gain.value = 0
   168→      hg.connect(master)
   169→      this.heartGain = hg
   170→
   171→      this.buildAmbience()
   172→
   173→      this.ready = true
   174→      this.loadRoverSample()
   175→      this.loadBgm()
   176→      this.loadMachinery()
   177→      this.apPlanet = null // force timbre application
   178→      this.applyAmbience()
   179→      if (this.heartGain) {
   180→        this.heartGain.gain.setTargetAtTime(clamp01(this.heartLevel) * 0.35, ctx.currentTime, 0.3)
   181→      }
   182→      // re-apply any loop state requested before init
   183→      if (this.burnOn) {
   184→        this.burnOn = false
   185→        this.landingBurn(true)
   186→      }
   187→      if (this.roverOn) {
   188→        const th = this.roverThrottle
   189→        this.roverOn = false
   190→        this.setRoverEngine(true, th)
   191→      }
   192→      if (this.motorOn) {
   193→        const th = this.motorThrottle
   194→        this.motorOn = false
   195→        this.setConstructionMotor(true, th)
   196→      }
   197→
   198→      if (ctx.state === 'suspended') ctx.resume().catch(() => {})
   199→      if (!this.schedTimer) {
   200→        this.schedTimer = setInterval(() => this.schedTick(), 200)
   201→      }
   202→    } catch {
   203→      // never leave a half-built engine marked ready
   204→      this.ctx = null
   205→      this.master = null
   206→      this.ready = false
   207→    }
   208→  }
   209→
   210→  resume(): void {
   211→    const ctx = this.ctx
   212→    if (ctx && ctx.state === 'suspended') ctx.resume().catch(() => {})
   213→  }
   214→
   215→  suspend(): void {
   216→    const ctx = this.ctx
   217→    if (ctx && ctx.state === 'running') ctx.suspend().catch(() => {})
   218→  }
   219→
   220→  setMasterVolume(v: number): void {
   221→    const vol = clamp01(v)
   222→    this.volume = vol
   223→    if (!this.ready || !this.master || !this.ctx) return
   224→    try {
   225→      this.master.gain.setTargetAtTime(vol, this.ctx.currentTime, 0.05)
   226→    } catch {
   227→      /* ignore */
   228→    }
   229→  }
   230→
   231→  setDoppler(on: boolean): void {
   232→    this.doppler = !!on
   233→  }
   234→
   235→  setListener(pos: AudioPosition, forward: { x: number; y: number; z: number }): void {
   236→    const p = pos ?? { x: 0, y: 1.6, z: 0 }
   237→    this.listenerPos = { x: num(p.x, 0), y: num(p.y, 1.6), z: num(p.z, 0) }
   238→    const f = forward ?? { x: 0, y: 0, z: -1 }
   239→    let fx = num(f.x, 0)
   240→    let fy = num(f.y, 0)
   241→    let fz = num(f.z, -1)
   242→    const m = Math.max(Math.abs(fx), Math.abs(fy), Math.abs(fz))
   243→    if (m < 1e-6) {
   244→      fx = 0
   245→      fy = 0
   246→      fz = -1
   247→    } else {
   248→      fx /= m
   249→      fy /= m
   250→      fz /= m
   251→    }
   252→    this.listenerFwd = { x: fx, y: fy, z: fz }
   253→
   254→    const ctx = this.ctx
   255→    if (!ctx || !this.ready) return
   256→    try {
   257→      const L = ctx.listener
   258→      const lp = this.listenerPos
   259→      if (L.positionX) {
   260→        L.positionX.value = lp.x
   261→        L.positionY.value = lp.y
   262→        L.positionZ.value = lp.z
   263→      } else {
   264→        L.setPosition(lp.x, lp.y, lp.z)
   265→      }
   266→      if (L.forwardX) {
   267→        L.forwardX.value = fx
   268→        L.forwardY.value = fy
   269→        L.forwardZ.value = fz
   270→      } else {
   271→        L.setOrientation(fx, fy, fz, 0, 1, 0)
   272→      }
   273→    } catch {
   274→      /* ignore */
   275→    }
   276→  }

   299→  setFan(intensity: number): void {
   300→    const lvl = clamp01(num(intensity, 0.4))
   301→    if (Math.abs(lvl - this.fanLevel) < 0.01) return
   302→    this.fanLevel = lvl
   303→    if (!this.ready) return
   304→    this.applyAmbience()
   305→  }
   306→
   307→  setAlarm(on: boolean): void {
   308→    const next = !!on
   309→    if (next === this.alarmOn) return
   310→    this.alarmOn = next
   311→    if (!this.ready) return
   312→    const ctx = this.ctx
   313→    if (!ctx || ctx.state !== 'running') return
   314→    if (next) {
   315→      // immediate first beep, then the scheduler keeps the pattern going
   316→      this.alarmBeep(ctx.currentTime + 0.08, this.alarmHi ? 880 : 660)
   317→      this.alarmHi = !this.alarmHi
   318→      this.nextAlarmAt = ctx.currentTime + 1.2
   319→    }
   320→  }
   321→
   322→  setBreathing(on: boolean, effort = 0.4): void {
   323→    const e = clamp01(num(effort, 0.4))
   324→    if (on === this.breathOn && Math.abs(e - this.breathEffort) < 0.02) return
   325→    this.breathOn = !!on
   326→    this.breathEffort = e
   327→    if (!this.ready) return
   328→    this.applyBreathing()
   329→  }
   330→
   331→  setHeartbeat(intensity = 0): void {
   332→    const lvl = clamp01(num(intensity, 0))
   333→    const wasOff = this.heartLevel <= 0.02
   334→    const changed = Math.abs(lvl - this.heartLevel) > 0.015
   335→    this.heartLevel = lvl
   336→    if (!this.ready) return
   337→    const ctx = this.ctx
   338→    if (!ctx || !this.heartGain) return
   339→    if (changed) {
   340→      try {
   341→        this.heartGain.gain.setTargetAtTime(lvl * 0.35, ctx.currentTime, 0.4)
   342→      } catch {
   343→        /* ignore */
   344→      }
   345→    }
   346→    if (wasOff && lvl > 0.02) this.nextHeartAt = 0
   347→  }
   348→
   349→  setRoverEngine(on: boolean, throttle = 0): void {
   350→    const th = clamp01(num(throttle, 0))
   351→    const changed = on !== this.roverOn || Math.abs(th - this.roverThrottle) >= 0.015
   352→    this.roverOn = !!on
   353→    this.roverThrottle = th
   354→    if (!this.ready) return
   355→    const ctx = this.ctx
   356→    if (!ctx) return
   357→    // prefer the real recorded rover audio once it has decoded
   358→    if (!this.roverBuf && !this.roverBufTried) this.loadRoverSample()
   359→    if (this.roverBuf) {
   360→      if (!changed && (on ? this.roverReal !== null : this.roverReal === null)) return
   361→      this.applyRoverReal(on, th)
   362→      return
   363→    }
   364→    if (!on) {
   365→      if (changed) this.killLoop('rover', 0.35)
   366→      return
   367→    }
   368→    if (!changed && this.loops.has('rover')) return
   369→    const loop = this.ensureLoop('rover', null, (out) => this.buildRover(out))
   370→    if (!loop) return
   371→    const t = ctx.currentTime
   372→    const f = 74 + th * 92
   373→    try {
   374→      loop.params.freq.setTargetAtTime(f, t, 0.15)
   375→      loop.params.sub.setTargetAtTime(f / 2, t, 0.15)
   376→      loop.params.lp.setTargetAtTime(340 + th * 1200, t, 0.2)
   377→      loop.params.eng.setTargetAtTime(0.045 + th * 0.1, t, 0.18)
   378→      loop.params.wheel.setTargetAtTime(th > 0.02 ? th * 0.11 : 0.0001, t, 0.25)
   379→      loop.ctrl.gain.setTargetAtTime(1, t, 0.2)
   380→    } catch {
   381→      /* ignore */
   382→    }
   383→  }

   385→  // --------------------------------------------------------------
   386→  // Real recorded rover audio (public NASA/JPL Perseverance drive
   387→  // recording, Sol 16 — wheels + motors on Mars regolith).
   388→  // --------------------------------------------------------------
   389→  private loadRoverSample(): void {
   390→    if (this.roverBufTried) return
   391→    if (typeof window === 'undefined' || typeof fetch === 'undefined') return
   392→    const ctx = this.ctx
   393→    if (!ctx) return
   394→    this.roverBufTried = true
   395→    fetch('/sounds/rover-drive.mp3')
   396→      .then((r) => (r.ok ? r.arrayBuffer() : Promise.reject(new Error(String(r.status)))))
   397→      .then((ab) => ctx.decodeAudioData(ab))
   398→      .then((buf) => {
   399→        this.roverBuf = buf
   400→        if (this.roverOn && this.ready) this.setRoverEngine(true, this.roverThrottle)
   401→      })
   402→      .catch(() => {
   403→        // sample unavailable — procedural engine keeps running
   404→      })
   405→  }
   406→
   407→  private applyRoverReal(on: boolean, th: number): void {
   408→    const ctx = this.ctx
   409→    const master = this.master
   410→    if (!ctx || !master) return
   411→    const t = ctx.currentTime
   412→    if (this.loops.has('rover')) this.killLoop('rover', 0.2) // replace procedural fallback
   413→    if (!on) {
   414→      const r = this.roverReal
   415→      this.roverReal = null
   416→      if (r) {
   417→        try {
   418→          r.gain.gain.setTargetAtTime(0.0001, t, 0.12)
   419→          r.src.stop(t + 0.6)
   420→        } catch {
   421→          /* ignore */
   422→        }
   423→      }
   424→      return
   425→    }
   426→    let r = this.roverReal
   427→    if (!r) {
   428→      try {
   429→        const src = ctx.createBufferSource()
   430→        src.buffer = this.roverBuf!
   431→        src.loop = true
   432→        const filter = ctx.createBiquadFilter()
   433→        filter.type = 'lowpass'
   434→        filter.frequency.value = 1200
   435→        filter.Q.value = 0.4
   436→        const gain = ctx.createGain()
   437→        gain.gain.value = 0.0001
   438→        src.connect(filter)
   439→        filter.connect(gain)
   440→        gain.connect(master)
   441→        const dur = this.roverBuf?.duration ?? 1
   442→        src.start(0, Math.random() * Math.max(0.1, dur - 2))
   443→        r = this.roverReal = { src, gain, filter }
   444→      } catch {
   445→        return
   446→      }
   447→    }
   448→    try {
   449→      r.gain.gain.setTargetAtTime(0.3 + th * 0.6, t, 0.15)
   450→      r.filter.frequency.setTargetAtTime(1400 + th * 2600, t, 0.2)
   451→      r.src.playbackRate.setTargetAtTime(0.82 + th * 0.4, t, 0.2)
   452→    } catch {
   453→      /* ignore */
   454→    }
   455→  }

   457→  // --------------------------------------------------------------
   458→  // Background music — user-supplied ambient track (Breath on the
   459→  // Glass), looped quietly like an environmental sound bed.
   460→  // --------------------------------------------------------------
   461→  setBgm(on: boolean, volume = 0.16): void {
   462→    this.bgmOn = !!on
   463→    this.bgmVol = clamp01(num(volume, 0.16))
   464→    if (!this.ready) return
   465→    const ctx = this.ctx
   466→    if (!ctx || !this.master) return
   467→    if (!this.bgmBuf && !this.bgmTried) this.loadBgm()
   468→    if (!on || !this.bgmBuf) return
   469→    if (this.bgmReal) {
   470→      try {
   471→        this.bgmReal.gain.gain.setTargetAtTime(this.bgmVol, ctx.currentTime, 0.8)
   472→      } catch {
   473→        /* ignore */
   474→      }
   475→      return
   476→    }
   477→    try {
   478→      const src = ctx.createBufferSource()
   479→      src.buffer = this.bgmBuf
   480→      src.loop = true
   481→      const lp = ctx.createBiquadFilter()
   482→      lp.type = 'lowpass'
   483→      lp.frequency.value = 9000
   484→      const gain = ctx.createGain()
   485→      gain.gain.value = 0.0001
   486→      src.connect(lp)
   487→      lp.connect(gain)
   488→      gain.connect(this.master)
   489→      src.start(0, Math.random() * Math.max(0.1, (this.bgmBuf.duration ?? 1) - 2))
   490→      this.bgmReal = { src, gain }
   491→      gain.gain.setTargetAtTime(this.bgmVol, ctx.currentTime, 1.2)
   492→    } catch {
   493→      /* ignore */
   494→    }
   495→  }
   496→
   497→  private loadBgm(): void {
   498→    if (this.bgmTried) return
   499→    if (typeof window === 'undefined' || typeof fetch === 'undefined') return
   500→    const ctx = this.ctx
   501→    if (!ctx) return
   502→    this.bgmTried = true
   503→    fetch('/sounds/bgm-breath-on-the-glass.mp3')
   504→      .then((r) => (r.ok ? r.arrayBuffer() : Promise.reject(new Error(String(r.status)))))
   505→      .then((ab) => ctx.decodeAudioData(ab))
   506→      .then((buf) => {
   507→        this.bgmBuf = buf
   508→        if (this.bgmOn && this.ready) this.setBgm(true, this.bgmVol)
   509→      })
   510→      .catch(() => {
   511→        // sample unavailable — silent background
   512→      })
   513→  }

   515→  // --------------------------------------------------------------
   516→  // Colony machinery one-shots — real recorded industrial sounds
   517→  // that play at random intervals around the player (generators,
   518→  // pumps, compressors, servos working somewhere on the planet).
   519→  // --------------------------------------------------------------
   520→  playMachinery(volume = 0.12): void {
   521→    const v = clamp01(num(volume, 0.12))
   522→    const ctx = this.ctx
   523→    if (!ctx || !this.master || this.machBufs.length === 0 || v <= 0.001) return
   524→    if (this.shots >= this.SHOT_CAP) return
   525→    try {
   526→      const buf = this.machBufs[Math.floor(Math.random() * this.machBufs.length)]
   527→      const src = ctx.createBufferSource()
   528→      src.buffer = buf
   529→      src.playbackRate.value = 0.85 + Math.random() * 0.3
   530→      const lp = ctx.createBiquadFilter()
   531→      lp.type = 'lowpass'
   532→      lp.frequency.value = 2400 + Math.random() * 3200
   533→      const gain = ctx.createGain()
   534→      gain.gain.value = 0.0001
   535→      src.connect(lp)
   536→      lp.connect(gain)
   537→      const pan = ctx.createStereoPanner ? ctx.createStereoPanner() : null
   538→      let tail: AudioNode = gain
   539→      if (pan) {
   540→        pan.pan.value = Math.random() * 1.6 - 0.8
   541→        gain.connect(pan)
   542→        pan.connect(this.master)
   543→        tail = pan
   544→      } else {
   545→        gain.connect(this.master)
   546→      }
   547→      const dur = Math.min(buf.duration ?? 8, 6 + Math.random() * 9)
   548→      const off = Math.random() * Math.max(0.1, (buf.duration ?? 1) - dur - 0.2)
   549→      const t = ctx.currentTime + 0.03
   550→      gain.gain.setValueAtTime(0.0001, t)
   551→      gain.gain.setTargetAtTime(v, t, 1.1)
   552→      gain.gain.setTargetAtTime(0.0001, t + Math.max(0.5, dur - 1.6), 0.55)
   553→      src.start(t, off)
   554→      src.stop(t + dur + 0.4)
   555→      this.shots++
   556→      src.onended = () => {
   557→        this.shots--
   558→        try { tail.disconnect() } catch { /* ignore */ }
   559→      }
   560→    } catch {
   561→      /* ignore */
   562→    }
   563→  }
   564→
   565→  machineryCount(): number {
   566→    return this.machBufs.length
   567→  }

   595→  setConstructionMotor(on: boolean, throttle = 0): void {
   596→    const th = clamp01(num(throttle, 0))
   597→    const changed = on !== this.motorOn || Math.abs(th - this.motorThrottle) >= 0.02
   598→    this.motorOn = !!on
   599→    this.motorThrottle = th
   600→    if (!this.ready) return
   601→    const ctx = this.ctx
   602→    if (!ctx) return
   603→    if (!on) {
   604→      if (changed) this.killLoop('cmotor', 0.4)
   605→      return
   606→    }
   607→    if (!changed && this.loops.has('cmotor')) return
   608→    const loop = this.ensureLoop('cmotor', null, (out) => this.buildMotor(out))
   609→    if (!loop) return
   610→    const t = ctx.currentTime
   611→    try {
   612→      loop.params.servo.setTargetAtTime(520 + th * 780, t, 0.2)
   613→      loop.params.main.setTargetAtTime(0.35 + th * 0.65, t, 0.25)
   614→      loop.ctrl.gain.setTargetAtTime(1, t, 0.25)
   615→    } catch {
   616→      /* ignore */
   617→    }
   618→  }

  1239→  private makeNoiseBuffer(kind: 'white' | 'brown'): AudioBuffer {
  1240→    const ctx = this.ctx
  1241→    if (!ctx) throw new Error('no context')
  1242→    const len = Math.floor(ctx.sampleRate * 2)
  1243→    const buf = ctx.createBuffer(1, len, ctx.sampleRate)
  1244→    const d = buf.getChannelData(0)
  1245→    if (kind === 'white') {
  1246→      for (let i = 0; i < len; i++) d[i] = Math.random() * 2 - 1
  1247→    } else {
  1248→      // brown noise: integrated white, normalized
  1249→      let last = 0
  1250→      let peak = 0
  1251→      for (let i = 0; i < len; i++) {
  1252→        const w = Math.random() * 2 - 1
  1253→        last = (last + 0.02 * w) / 1.02
  1254→        d[i] = last
  1255→        const a = Math.abs(last)
  1256→        if (a > peak) peak = a
  1257→      }
  1258→      if (peak > 1e-6) {
  1259→        const k = 0.9 / peak
  1260→        for (let i = 0; i < len; i++) d[i] *= k
  1261→      }
  1262→    }
  1263→    return buf
  1264→  }

  1366→  private applyAmbience(): void {
  1367→    const ctx = this.ctx
  1368→    if (!ctx || !this.ready) return
  1369→    const t = ctx.currentTime
  1370→    const { inside, planet, windIntensity, storm } = this.env
  1371→    const wi = windIntensity
  1372→
  1373→    // planet timbre (rare change)
  1374→    if (this.windFilter && this.windWhiteG && this.windBrownG && this.apPlanet !== planet) {
  1375→      this.apPlanet = planet
  1376→      try {
  1377→        if (planet === 'moon') {
  1378→          // thin / airy
  1379→          this.windFilter.type = 'bandpass'
  1380→          this.windFilter.frequency.setTargetAtTime(1050, t, 0.4)
  1381→          this.windFilter.Q.setTargetAtTime(0.55, t, 0.4)
  1382→          this.windWhiteG.gain.setTargetAtTime(0.9, t, 0.5)
  1383→          this.windBrownG.gain.setTargetAtTime(0.1, t, 0.5)
  1384→        } else {
  1385→          // lower / rumblier
  1386→          this.windFilter.type = 'lowpass'
  1387→          this.windFilter.frequency.setTargetAtTime(470, t, 0.4)
  1388→          this.windFilter.Q.setTargetAtTime(0.9, t, 0.4)
  1389→          this.windWhiteG.gain.setTargetAtTime(0.3, t, 0.5)
  1390→          this.windBrownG.gain.setTargetAtTime(0.65, t, 0.5)
  1391→        }
  1392→      } catch {
  1393→        /* ignore */
  1394→      }
  1395→    }
  1396→
  1397→    // wind level (muffled + reduced when inside; storms raise it a lot)
  1398→    let wind = inside ? 0.05 + wi * 0.03 : 0.05 + wi * 0.2
  1399→    if (planet === 'moon') wind *= 0.85
  1400→    if (storm) wind = inside ? Math.max(wind, 0.15) : Math.min(0.55, wind * 2.3 + 0.13)
  1401→    if (this.windGain && Math.abs(wind - this.apWind) > 0.003) {
  1402→      this.apWind = wind
  1403→      try {
  1404→        this.windGain.gain.setTargetAtTime(wind, t, inside ? 0.4 : 0.9)
  1405→      } catch {
  1406→        /* ignore */
  1407→      }
  1408→    }
  1409→
  1410→    // storm rattle layer
  1411→    const stormLvl = storm ? (inside ? 0.13 : 0.32 + wi * 0.2) : 0
  1412→    if (this.stormGain && Math.abs(stormLvl - this.apStorm) > 0.003) {
  1413→      this.apStorm = stormLvl
  1414→      try {
  1415→        this.stormGain.gain.setTargetAtTime(stormLvl, t, 1.1)
  1416→      } catch {
  1417→        /* ignore */
  1418→      }
  1419→    }
  1420→
  1421→    // room tone crossfade
  1422→    const room = inside ? 1 : 0
  1423→    if (this.roomGain && room !== this.apRoom) {
  1424→      this.apRoom = room
  1425→      try {
  1426→        this.roomGain.gain.setTargetAtTime(room, t, 0.5)
  1427→      } catch {
  1428→        /* ignore */
  1429→      }
  1430→    }
  1431→
  1432→    // fan crossfade (only audible inside)
  1433→    const fan = inside ? clamp01(this.fanLevel) * 0.5 : 0
  1434→    if (this.fanGain && Math.abs(fan - this.apFan) > 0.003) {
  1435→      this.apFan = fan
  1436→      try {
  1437→        this.fanGain.gain.setTargetAtTime(fan, t, 0.45)
  1438→      } catch {
  1439→        /* ignore */
  1440→      }
  1441→    }
  1442→
  1443→    // breathing is automatically silenced inside the suit-off interior
  1444→    this.applyBreathing()
  1445→  }

  1447→  private applyBreathing(): void {
  1448→    const ctx = this.ctx

  1454→      const loop = this.ensureLoop('breath', null, (out) => this.buildBreath(out))
  1455→      if (!loop) return

  1457→      const rate = 0.24 + effort * 0.26 // ~0.25Hz at rest -> ~0.5Hz under effort
  1458→      if (Math.abs(vol - this.apBreath) > 0.01) {
  1459→        this.apBreath = vol

  1461→          loop.ctrl.gain.setTargetAtTime(vol, t, 0.5)

  1463→          /* ignore */

  1469→          loop.params.rate.setTargetAtTime(rate, t, 0.6)

  1483→  private buildBreath(out: GainNode): LoopParts | null {
  1484→    const ctx = this.ctx
  1485→    if (!ctx) return null
  1486→    const src = this.noise(true, 'white')
  1487→    const bp = this.bq('bandpass', 430, 1.0)
  1488→    const env = this.gn(0)
  1489→    src.connect(bp)
  1490→    bp.connect(env)
  1491→    env.connect(out)
  1492→    // rectified LFO gives alternating inhale/exhale bursts
  1493→    const lfo = this.osc('sine', 0.25)
  1494→    const shaper = ctx.createWaveShaper()
  1495→    const n = 129
  1496→    const curve = new Float32Array(n)
  1497→    for (let i = 0; i < n; i++) {
  1498→      const x = (i / (n - 1)) * 2 - 1
  1499→      curve[i] = Math.abs(x)
  1500→    }
  1501→    shaper.curve = curve
  1502→    const depth = this.gn(0.55)
  1503→    lfo.connect(shaper)
  1504→    shaper.connect(depth)
  1505→    depth.connect(env.gain)
  1506→    // inhale brighter / exhale darker
  1507→    const fd = this.gn(115)
  1508→    lfo.connect(fd)
  1509→    fd.connect(bp.frequency)
  1510→    src.start()
  1511→    lfo.start()
  1512→    return {
  1513→      sources: [src, lfo],

  1638→  // ---------- loop infrastructure ----------
  1639→
  1640→  private ensureLoop(
  1641→    id: string,
  1642→    pos: AudioPosition | null,
  1643→    build: (out: GainNode) => LoopParts | null
  1644→  ): LoopNode | null {
  1645→    const existing = this.loops.get(id)
  1646→    if (existing) return existing
  1647→    const ctx = this.ctx
  1648→    const master = this.master
  1649→    if (!ctx || !master) return null
  1650→    try {
  1651→      const ctrl = ctx.createGain()
  1652→      ctrl.gain.value = 0
  1653→      let panner: PannerNode | null = null
  1654→      if (pos) panner = this.makePanner(pos)
  1655→      if (panner) {
  1656→        ctrl.connect(panner)
  1657→        panner.connect(master)
  1658→      } else {
  1659→        ctrl.connect(master)
  1660→      }
  1661→      const parts = build(ctrl)
  1662→      if (!parts) {
  1663→        try {
  1664→          ctrl.disconnect()
  1665→          if (panner) panner.disconnect()
  1666→        } catch {
  1667→          /* ignore */
  1668→        }
  1669→        return null
  1670→      }
  1671→      const loop: LoopNode = {
  1672→        id,
  1673→        ctrl,
  1674→        panner,
  1675→        sources: parts.sources,
  1676→        cleanup: [ctrl, ...(panner ? [panner] : []), ...parts.cleanup],
  1677→        params: parts.params,
  1678→      }
  1679→      this.loops.set(id, loop)
  1680→      return loop
  1681→    } catch {
  1682→      return null
  1683→    }
  1684→  }
  1685→
  1686→  private killLoop(id: string, fade: number): void {
  1687→    const loop = this.loops.get(id)
  1688→    if (!loop) return
  1689→    this.loops.delete(id)
  1690→    const ctx = this.ctx
  1691→    if (!ctx) return
  1692→    const t = ctx.currentTime
  1693→    const f = Math.max(0.05, fade)
  1694→    try {
  1695→      loop.ctrl.gain.setTargetAtTime(0.0001, t, f / 3)
  1696→    } catch {
  1697→      /* ignore */
  1698→    }
  1699→    const stopAt = t + f + 0.15
  1700→    let stopped = false
  1701→    const dispose = (): void => {
  1702→      if (stopped) return
  1703→      stopped = true
  1704→      for (const node of loop.cleanup) {
  1705→        try {
  1706→          node.disconnect()
  1707→        } catch {
  1708→          /* ignore */
  1709→        }
  1710→      }
  1711→    }
  1712→    for (const s of loop.sources) {
  1713→      s.onended = dispose
  1714→      try {
  1715→        s.stop(stopAt)
  1716→      } catch {
  1717→        /* ignore */
  1718→      }
  1719→    }
  1720→  }

  1724→  private makePanner(pos: AudioPosition): PannerNode | null {
  1725→    const ctx = this.ctx
  1726→    if (!ctx) return null
  1727→    try {
  1728→      const p = ctx.createPanner()
  1729→      p.panningModel = 'equalpower'
  1730→      p.distanceModel = 'inverse'
  1731→      p.refDistance = 3
  1732→      p.rolloffFactor = 1.2
  1733→      p.maxDistance = 10000
  1734→      if (this.doppler) {
  1735→        try {
  1736→          // legacy knob, removed from modern spec typings — best effort only
  1737→          const legacy = p as unknown as { dopplerFactor?: number }
  1738→          legacy.dopplerFactor = 1
  1739→        } catch {
  1740→          /* not supported — fine */
  1741→        }
  1742→      }
  1743→      this.setNodePos(p, pos)
  1744→      return p
  1745→    } catch {
  1746→      return null
  1747→    }
  1748→  }

  1750→  private setNodePos(p: PannerNode, pos: AudioPosition): void {
  1751→    try {
  1752→      if (p.positionX) {
  1753→        p.positionX.value = pos.x
  1754→        p.positionY.value = pos.y
  1755→        p.positionZ.value = pos.z
  1756→      } else {
  1757→        p.setPosition(pos.x, pos.y, pos.z)
  1758→      }
  1759→    } catch {
  1760→      /* ignore */
  1761→    }
  1762→  }

  1764→  // ---------- one-shot infrastructure ----------
  1765→
  1766→  /** Gate + try/catch wrapper for fire-and-forget one-shots. */
  1767→  private guard(fn: (ctx: AudioContext, master: GainNode) => void): void {
  1768→    const ctx = this.ctx
  1769→    const master = this.master
  1770→    if (!this.ready || !ctx || !master) return
  1771→    if (this.shots >= this.SHOT_CAP) return
  1772→    try {
  1773→      fn(ctx, master)
  1774→    } catch {
  1775→      /* audio must never throw into the game loop */
  1776→    }
  1777→  }
  1778→
  1779→  /** Registers one-shot sources for bounded lifetime + node cleanup. */
  1780→  private trackShot(sources: AudioScheduledSourceNode[], cleanup: AudioNode[], stopAt: number): void {
  1781→    this.shots++
  1782→    let left = sources.length
  1783→    const done = (): void => {
  1784→      left -= 1
  1785→      if (left > 0) return
  1786→      this.shots = Math.max(0, this.shots - 1)
  1787→      for (const n of cleanup) {
  1788→        try {
  1789→          n.disconnect()
  1790→        } catch {
  1791→          /* ignore */
  1792→        }
  1793→      }
  1794→    }
  1795→    for (const s of sources) {
  1796→      s.onended = done
  1797→      try {
  1798→        s.stop(stopAt)
  1799→      } catch {
  1800→        /* ignore */
  1801→      }
  1802→    }
  1803→  }

  1837→  /** Generic short tone with soft envelope; optional exponential glide. */
  1838→  private tone(
  1839→    t: number,
  1840→    freq: number,
  1841→    dur: number,
  1842→    vol: number,
  1843→    type: OscillatorType = 'sine',
  1844→    glideTo = 0
  1845→  ): void {
  1846→    const ctx = this.ctx
  1847→    const master = this.master
  1848→    if (!ctx || !master) return
  1849→    const o = this.osc(type, freq)
  1850→    const g = this.gn(0.0001)
  1851→    o.connect(g)
  1852→    g.connect(master)
  1853→    g.gain.setValueAtTime(0.0001, t)
  1854→    g.gain.linearRampToValueAtTime(vol, t + 0.018)
  1855→    g.gain.setTargetAtTime(0.0001, t + dur * 0.55, Math.max(0.02, dur * 0.16))
  1856→    if (glideTo > 0) {
  1857→      o.frequency.setValueAtTime(Math.max(1, freq), t)
  1858→      o.frequency.exponentialRampToValueAtTime(glideTo, t + dur * 0.9)
  1859→    }
  1860→    o.start(t)
  1861→    this.trackShot([o], [o, g], t + dur + 0.4)
  1862→  }

  1864→  /** Tiny filtered-noise tick (squelch clicks, latch clicks). */
  1865→  private click(t: number, freq: number, vol: number): void {
  1866→    const ctx = this.ctx
  1867→    const master = this.master
  1868→    if (!ctx || !master) return
  1869→    const src = this.noise(true, 'white')
  1870→    const bp = this.bq('bandpass', freq, 1.4)
  1871→    const g = this.gn(0.0001)
  1872→    src.connect(bp)
  1873→    bp.connect(g)
  1874→    g.connect(master)
  1875→    g.gain.setValueAtTime(0.0001, t)
  1876→    g.gain.linearRampToValueAtTime(vol, t + 0.006)
  1877→    g.gain.setTargetAtTime(0.0001, t + 0.008, 0.012)
  1878→    src.start(t)
  1879→    this.trackShot([src], [src, bp, g], t + 0.18)
  1880→  }

  1884→    const ctx = this.ctx
  1885→    const master = this.master
  1886→    if (!ctx || !master) return
  1887→    const o = this.osc('sine', 130)
  1888→    const g = this.gn(0.0001)

  1890→    g.connect(master)

  1893→    g.gain.setValueAtTime(0.0001, t)
  1894→    g.gain.linearRampToValueAtTime(vol, t + 0.006)

  1900→  // ---------- scheduler (alarm + heartbeat) ----------
  1901→
  1902→  private schedTick(): void {
  1903→    const ctx = this.ctx
  1904→    if (!ctx || ctx.state !== 'running') return
  1905→    const t = ctx.currentTime
  1906→    if (this.alarmOn) {
  1907→      if (t >= this.nextAlarmAt) {
  1908→        this.alarmBeep(t + 0.06, this.alarmHi ? 880 : 660)
  1909→        this.alarmHi = !this.alarmHi
  1910→        this.nextAlarmAt = t + 1.2
  1911→      }
  1912→    } else {
  1913→      this.nextAlarmAt = 0
  1914→    }
  1915→    if (this.heartLevel > 0.02) {
  1916→      if (t >= this.nextHeartAt) {
  1917→        this.heartBeat(t + 0.06)
  1918→        const bpm = 60 + clamp01(this.heartLevel) * 72
  1919→        this.nextHeartAt = t + 60 / bpm
  1920→      }
  1921→    } else {
  1922→      this.nextHeartAt = 0
  1923→    }
  1924→  }
  1925→
  1926→  private alarmBeep(t: number, freq: number): void {
  1927→    const ctx = this.ctx
  1928→    const master = this.master
  1929→    if (!ctx || !master) return
  1930→    try {
  1931→      const o = this.osc('sine', freq)
  1932→      const h = this.osc('sine', freq * 2)
  1933→      const hg = this.gn(0.22)
  1934→      const g = this.gn(0.0001)
  1935→      o.connect(g)
  1936→      h.connect(hg)
  1937→      hg.connect(g)
  1938→      g.connect(master)
  1939→      g.gain.setValueAtTime(0.0001, t)
  1940→      g.gain.linearRampToValueAtTime(0.11, t + 0.014)
  1941→      g.gain.setValueAtTime(0.11, t + 0.1)
  1942→      g.gain.setTargetAtTime(0.0001, t + 0.11, 0.04)
  1943→      o.start(t)
  1944→      h.start(t)
  1945→      this.trackShot([o, h], [o, h, g, hg], t + 0.4)
  1946→    } catch {
  1947→      /* ignore */
  1948→    }
  1949→  }
  1950→
  1951→  private heartBeat(t: number): void {
  1952→    const ctx = this.ctx
  1953→    const bus = this.heartGain
  1954→    if (!ctx || !bus) return
  1955→    try {
  1956→      const oscs: OscillatorNode[] = []
  1957→      const nodes: AudioNode[] = []
  1958→      const pair = (dt: number, amp: number, f0: number): void => {
  1959→        const o = ctx.createOscillator()
  1960→        o.type = 'sine'
  1961→        o.frequency.setValueAtTime(f0, t + dt)
  1962→        o.frequency.exponentialRampToValueAtTime(f0 * 0.72, t + dt + 0.13)
  1963→        const g = ctx.createGain()
  1964→        g.gain.setValueAtTime(0.0001, t + dt)
  1965→        g.gain.linearRampToValueAtTime(amp, t + dt + 0.02)
  1966→        g.gain.setTargetAtTime(0.0001, t + dt + 0.035, 0.05)
  1967→        o.connect(g)
  1968→        g.connect(bus)
  1969→        o.start(t + dt)
  1970→        oscs.push(o)
  1971→        nodes.push(o, g)
  1972→      }
  1973→      pair(0, 1, 52) // lub
  1974→      pair(0.21, 0.62, 44) // dub
  1975→      this.trackShot(oscs, nodes, t + 0.75)
  1976→    } catch {
  1977→      /* ignore */
  1978→    }
  1979→  }
  1980→}
