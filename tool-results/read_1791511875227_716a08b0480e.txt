     1→// ============================================================
     2→// Procedural audio engine — Off-World Colony STEM Game (Task 3-a)
     3→// 100% synthesized with Web Audio API. No files, no fetch, no assets.
     4→//
     5→// Design:
     6→//  - one shared AudioContext; master chain: masterGain -> compressor -> destination
     7→//  - two shared 2s noise buffers (white + brown) reused by every noise sound
     8→//  - persistent ambience layers (wind / storm / room tone / fan) crossfaded
     9→//    with setTargetAtTime — never hard switches
    10→//  - every public method is safe to call any time (no-ops before init()),
    11→//    cheap when called every frame (state caching + epsilon gates)
    12→//  - one-shots capped (~24 concurrent), auto-cleanup via onended
    13→// ============================================================
    14→
    15→export type AudioPosition = { x: number; y: number; z: number }
    16→export type SurfaceType = 'regolith' | 'metal'
    17→export type Planet = 'moon' | 'mars'
    18→
    19→// ---------- tiny helpers ----------
    20→
    21→const clamp01 = (v: number): number => {
    22→  if (typeof v !== 'number' || !isFinite(v)) return 0
    23→  return v < 0 ? 0 : v > 1 ? 1 : v
    24→}
    25→
    26→const num = (v: number, d: number): number =>
    27→  typeof v === 'number' && isFinite(v) ? v : d
    28→
    29→interface LoopParts {
    30→  sources: AudioScheduledSourceNode[]
    31→  cleanup: AudioNode[]
    32→  params: Record<string, AudioParam>
    33→}
    34→
    35→interface LoopNode extends LoopParts {
    36→  id: string
    37→  ctrl: GainNode // fade / master control gain for the loop
    38→  panner: PannerNode | null
    39→}
    40→
    41→// real recorded machinery/industrial clips in public/sounds/machinery
    42→// (generators, pumps, compressors, ventilation, hydraulics — downloaded
    43→// from the internet; see tools/fetch-sounds.sh)

    45→  'generator.mp3',

   139→  init(): void {
   140→    if (this.ready) {
   141→      this.resume()
   142→      return
   143→    }
   144→    if (typeof window === 'undefined') return
   145→    try {
   146→      const w = window as unknown as {
   147→        AudioContext?: typeof AudioContext
   148→        webkitAudioContext?: typeof AudioContext
   149→      }
   150→      const Ctor = w.AudioContext ?? w.webkitAudioContext
   151→      if (!Ctor) return
   152→
   153→      const ctx = new Ctor()
   154→      this.ctx = ctx
   155→
   156→      // shared noise buffers (2s each)
   157→      this.white = this.makeNoiseBuffer('white')
   158→      this.brown = this.makeNoiseBuffer('brown')
   159→
   160→      // master chain: gain -> compressor -> destination
   161→      const master = ctx.createGain()
   162→      master.gain.value = clamp01(this.volume)
   163→      const comp = ctx.createDynamicsCompressor()
   164→      comp.threshold.value = -16
   165→      comp.knee.value = 18
   166→      comp.ratio.value = 5
   167→      comp.attack.value = 0.004
   168→      comp.release.value = 0.24
   169→      master.connect(comp)
   170→      comp.connect(ctx.destination)
   171→      this.master = master
   172→
   173→      // heartbeat bus
   174→      const hg = ctx.createGain()
   175→      hg.gain.value = 0
   176→      hg.connect(master)
   177→      this.heartGain = hg
   178→
   179→      this.buildAmbience()
   180→
   181→      this.ready = true
   182→      this.loadRoverSample()
   183→      this.loadBgm()
   184→      this.loadMachinery()
   185→      this.loadCutsceneChatter() // warm the landing voice track early
   186→      this.apPlanet = null // force timbre application
   187→      this.applyAmbience()
   188→      if (this.heartGain) {
   189→        this.heartGain.gain.setTargetAtTime(clamp01(this.heartLevel) * 0.35, ctx.currentTime, 0.3)
   190→      }
   191→      // re-apply any loop state requested before init
   192→      if (this.burnOn) {
   193→        this.burnOn = false
   194→        this.landingBurn(true)
   195→      }
   196→      if (this.roverOn) {
   197→        const th = this.roverThrottle
   198→        this.roverOn = false
   199→        this.setRoverEngine(true, th)
   200→      }
   201→      if (this.motorOn) {
   202→        const th = this.motorThrottle
   203→        this.motorOn = false
   204→        this.setConstructionMotor(true, th)
   205→      }
   206→
   207→      if (ctx.state === 'suspended') ctx.resume().catch(() => {})
   208→      if (!this.schedTimer) {
   209→        this.schedTimer = setInterval(() => this.schedTick(), 200)
   210→      }
   211→    } catch {
   212→      // never leave a half-built engine marked ready
   213→      this.ctx = null
   214→      this.master = null
   215→      this.ready = false
   216→    }
   217→  }
   218→
   219→  resume(): void {
   220→    const ctx = this.ctx
   221→    if (ctx && ctx.state === 'suspended') ctx.resume().catch(() => {})
   222→  }
   223→
   224→  suspend(): void {
   225→    const ctx = this.ctx
   226→    if (ctx && ctx.state === 'running') ctx.suspend().catch(() => {})
   227→  }
   228→
   229→  setMasterVolume(v: number): void {
   230→    const vol = clamp01(v)
   231→    this.volume = vol
   232→    if (!this.ready || !this.master || !this.ctx) return
   233→    try {
   234→      this.master.gain.setTargetAtTime(vol, this.ctx.currentTime, 0.05)
   235→    } catch {
   236→      /* ignore */
   237→    }
   238→  }
   239→
   240→  setDoppler(on: boolean): void {
   241→    this.doppler = !!on
   242→  }
   243→
   244→  setListener(pos: AudioPosition, forward: { x: number; y: number; z: number }): void {
   245→    const p = pos ?? { x: 0, y: 1.6, z: 0 }
   246→    this.listenerPos = { x: num(p.x, 0), y: num(p.y, 1.6), z: num(p.z, 0) }
   247→    const f = forward ?? { x: 0, y: 0, z: -1 }
   248→    let fx = num(f.x, 0)
   249→    let fy = num(f.y, 0)
   250→    let fz = num(f.z, -1)
   251→    const m = Math.max(Math.abs(fx), Math.abs(fy), Math.abs(fz))
   252→    if (m < 1e-6) {
   253→      fx = 0
   254→      fy = 0
   255→      fz = -1
   256→    } else {
   257→      fx /= m
   258→      fy /= m
   259→      fz /= m
   260→    }
   261→    this.listenerFwd = { x: fx, y: fy, z: fz }
   262→
   263→    const ctx = this.ctx
   264→    if (!ctx || !this.ready) return
   265→    try {
   266→      const L = ctx.listener
   267→      const lp = this.listenerPos
   268→      if (L.positionX) {
   269→        L.positionX.value = lp.x
   270→        L.positionY.value = lp.y
   271→        L.positionZ.value = lp.z
   272→      } else {
   273→        L.setPosition(lp.x, lp.y, lp.z)
   274→      }
   275→      if (L.forwardX) {
   276→        L.forwardX.value = fx
   277→        L.forwardY.value = fy
   278→        L.forwardZ.value = fz
   279→      } else {
   280→        L.setOrientation(fx, fy, fz, 0, 1, 0)
   281→      }
   282→    } catch {
   283→      /* ignore */
   284→    }
   285→  }

   308→  setFan(intensity: number): void {
   309→    const lvl = clamp01(num(intensity, 0.4))
   310→    if (Math.abs(lvl - this.fanLevel) < 0.01) return
   311→    this.fanLevel = lvl
   312→    if (!this.ready) return
   313→    this.applyAmbience()
   314→  }
   315→
   316→  setAlarm(on: boolean): void {
   317→    const next = !!on
   318→    if (next === this.alarmOn) return
   319→    this.alarmOn = next
   320→    if (!this.ready) return
   321→    const ctx = this.ctx
   322→    if (!ctx || ctx.state !== 'running') return
   323→    if (next) {
   324→      // immediate first beep, then the scheduler keeps the pattern going
   325→      this.alarmBeep(ctx.currentTime + 0.08, this.alarmHi ? 880 : 660)
   326→      this.alarmHi = !this.alarmHi
   327→      this.nextAlarmAt = ctx.currentTime + 1.2
   328→    }
   329→  }
   330→
   331→  setBreathing(on: boolean, effort = 0.4): void {
   332→    const e = clamp01(num(effort, 0.4))
   333→    if (on === this.breathOn && Math.abs(e - this.breathEffort) < 0.02) return
   334→    this.breathOn = !!on
   335→    this.breathEffort = e
   336→    if (!this.ready) return
   337→    this.applyBreathing()
   338→  }
   339→
   340→  setHeartbeat(intensity = 0): void {
   341→    const lvl = clamp01(num(intensity, 0))
   342→    const wasOff = this.heartLevel <= 0.02
   343→    const changed = Math.abs(lvl - this.heartLevel) > 0.015
   344→    this.heartLevel = lvl
   345→    if (!this.ready) return
   346→    const ctx = this.ctx
   347→    if (!ctx || !this.heartGain) return
   348→    if (changed) {
   349→      try {
   350→        this.heartGain.gain.setTargetAtTime(lvl * 0.35, ctx.currentTime, 0.4)
   351→      } catch {
   352→        /* ignore */
   353→      }
   354→    }
   355→    if (wasOff && lvl > 0.02) this.nextHeartAt = 0
   356→  }
   357→
   358→  setRoverEngine(on: boolean, throttle = 0): void {
   359→    const th = clamp01(num(throttle, 0))
   360→    const changed = on !== this.roverOn || Math.abs(th - this.roverThrottle) >= 0.015
   361→    this.roverOn = !!on
   362→    this.roverThrottle = th
   363→    if (!this.ready) return
   364→    const ctx = this.ctx
   365→    if (!ctx) return
   366→    // prefer the real recorded rover audio once it has decoded
   367→    if (!this.roverBuf && !this.roverBufTried) this.loadRoverSample()
   368→    if (this.roverBuf) {
   369→      if (!changed && (on ? this.roverReal !== null : this.roverReal === null)) return
   370→      this.applyRoverReal(on, th)
   371→      return
   372→    }
   373→    if (!on) {
   374→      if (changed) this.killLoop('rover', 0.35)
   375→      return
   376→    }
   377→    if (!changed && this.loops.has('rover')) return
   378→    const loop = this.ensureLoop('rover', null, (out) => this.buildRover(out))
   379→    if (!loop) return
   380→    const t = ctx.currentTime
   381→    const f = 74 + th * 92
   382→    try {
   383→      loop.params.freq.setTargetAtTime(f, t, 0.15)
   384→      loop.params.sub.setTargetAtTime(f / 2, t, 0.15)
   385→      loop.params.lp.setTargetAtTime(340 + th * 1200, t, 0.2)
   386→      loop.params.eng.setTargetAtTime(0.045 + th * 0.1, t, 0.18)
   387→      loop.params.wheel.setTargetAtTime(th > 0.02 ? th * 0.11 : 0.0001, t, 0.25)
   388→      loop.ctrl.gain.setTargetAtTime(1, t, 0.2)
   389→    } catch {
   390→      /* ignore */
   391→    }
   392→  }

   394→  // --------------------------------------------------------------
   395→  // Real recorded rover audio (public NASA/JPL Perseverance drive
   396→  // recording, Sol 16 — wheels + motors on Mars regolith).
   397→  // --------------------------------------------------------------
   398→  private loadRoverSample(): void {
   399→    if (this.roverBufTried) return
   400→    if (typeof window === 'undefined' || typeof fetch === 'undefined') return
   401→    const ctx = this.ctx
   402→    if (!ctx) return
   403→    this.roverBufTried = true
   404→    fetch('/sounds/rover-drive.mp3')
   405→      .then((r) => (r.ok ? r.arrayBuffer() : Promise.reject(new Error(String(r.status)))))
   406→      .then((ab) => ctx.decodeAudioData(ab))
   407→      .then((buf) => {
   408→        this.roverBuf = buf
   409→        if (this.roverOn && this.ready) this.setRoverEngine(true, this.roverThrottle)
   410→      })
   411→      .catch(() => {
   412→        // sample unavailable — procedural engine keeps running
   413→      })
   414→  }
   415→
   416→  private applyRoverReal(on: boolean, th: number): void {
   417→    const ctx = this.ctx
   418→    const master = this.master
   419→    if (!ctx || !master) return
   420→    const t = ctx.currentTime
   421→    if (this.loops.has('rover')) this.killLoop('rover', 0.2) // replace procedural fallback
   422→    if (!on) {
   423→      const r = this.roverReal
   424→      this.roverReal = null
   425→      if (r) {
   426→        try {
   427→          r.gain.gain.setTargetAtTime(0.0001, t, 0.12)
   428→          r.src.stop(t + 0.6)
   429→        } catch {
   430→          /* ignore */
   431→        }
   432→      }
   433→      return
   434→    }
   435→    let r = this.roverReal
   436→    if (!r) {
   437→      try {
   438→        const src = ctx.createBufferSource()
   439→        src.buffer = this.roverBuf!
   440→        src.loop = true
   441→        const filter = ctx.createBiquadFilter()
   442→        filter.type = 'lowpass'
   443→        filter.frequency.value = 1200
   444→        filter.Q.value = 0.4
   445→        const gain = ctx.createGain()
   446→        gain.gain.value = 0.0001
   447→        src.connect(filter)
   448→        filter.connect(gain)
   449→        gain.connect(master)
   450→        const dur = this.roverBuf?.duration ?? 1
   451→        src.start(0, Math.random() * Math.max(0.1, dur - 2))
   452→        r = this.roverReal = { src, gain, filter }
   453→      } catch {
   454→        return
   455→      }
   456→    }
   457→    try {
   458→      r.gain.gain.setTargetAtTime(0.3 + th * 0.6, t, 0.15)
   459→      r.filter.frequency.setTargetAtTime(1400 + th * 2600, t, 0.2)
   460→      r.src.playbackRate.setTargetAtTime(0.82 + th * 0.4, t, 0.2)
   461→    } catch {
   462→      /* ignore */
   463→    }
   464→  }

   466→  // --------------------------------------------------------------
   467→  // Background music — user-supplied ambient track (Breath on the
   468→  // Glass), looped quietly like an environmental sound bed.
   469→  // --------------------------------------------------------------
   470→  setBgm(on: boolean, volume = 0.16): void {
   471→    this.bgmOn = !!on
   472→    this.bgmVol = clamp01(num(volume, 0.16))
   473→    if (!this.ready) return
   474→    const ctx = this.ctx
   475→    if (!ctx || !this.master) return
   476→    if (!this.bgmBuf && !this.bgmTried) this.loadBgm()
   477→    if (!on || !this.bgmBuf) return
   478→    if (this.bgmReal) {
   479→      try {
   480→        this.bgmReal.gain.gain.setTargetAtTime(this.bgmVol, ctx.currentTime, 0.8)
   481→      } catch {
   482→        /* ignore */
   483→      }
   484→      return
   485→    }
   486→    try {
   487→      const src = ctx.createBufferSource()
   488→      src.buffer = this.bgmBuf
   489→      src.loop = true
   490→      const lp = ctx.createBiquadFilter()
   491→      lp.type = 'lowpass'
   492→      lp.frequency.value = 9000
   493→      const gain = ctx.createGain()
   494→      gain.gain.value = 0.0001
   495→      src.connect(lp)
   496→      lp.connect(gain)
   497→      gain.connect(this.master)
   498→      src.start(0, Math.random() * Math.max(0.1, (this.bgmBuf.duration ?? 1) - 2))
   499→      this.bgmReal = { src, gain }
   500→      gain.gain.setTargetAtTime(this.bgmVol, ctx.currentTime, 1.2)
   501→    } catch {
   502→      /* ignore */
   503→    }
   504→  }
   505→
   506→  private loadBgm(): void {
   507→    if (this.bgmTried) return
   508→    if (typeof window === 'undefined' || typeof fetch === 'undefined') return
   509→    const ctx = this.ctx
   510→    if (!ctx) return
   511→    this.bgmTried = true
   512→    fetch('/sounds/bgm-breath-on-the-glass.mp3')
   513→      .then((r) => (r.ok ? r.arrayBuffer() : Promise.reject(new Error(String(r.status)))))
   514→      .then((ab) => ctx.decodeAudioData(ab))
   515→      .then((buf) => {
   516→        this.bgmBuf = buf
   517→        if (this.bgmOn && this.ready) this.setBgm(true, this.bgmVol)
   518→      })
   519→      .catch(() => {
   520→        // sample unavailable — silent background
   521→      })
   522→  }

   524→  // --------------------------------------------------------------
   525→  // Landing cinematic voice track — the user-supplied mission-control
   526→  // chatter ("cutscene audio"), played LOUD over the powered descent.
   527→  // One-shot from t=0; stopped/faded when the cinematic ends.
   528→  // --------------------------------------------------------------
   529→  setCutsceneChatter(on: boolean, volume = 0.95): void {
   530→    this.chatOn = !!on
   531→    this.chatVol = clamp01(num(volume, 0.95))
   532→    if (!this.ready) return
   533→    const ctx = this.ctx
   534→    if (!ctx || !this.master) return
   535→    if (!this.chatBuf && !this.chatTried) this.loadCutsceneChatter()
   536→    if (!on) {
   537→      if (this.chatReal) {
   538→        const r = this.chatReal
   539→        this.chatReal = null
   540→        try {
   541→          r.gain.gain.setTargetAtTime(0.0001, ctx.currentTime, 0.25)
   542→          r.src.stop(ctx.currentTime + 1.1)
   543→        } catch {
   544→          /* ignore */
   545→        }
   546→      }
   547→      return
   548→    }
   549→    if (!this.chatBuf || this.chatReal) return
   550→    try {
   551→      const src = ctx.createBufferSource()
   552→      src.buffer = this.chatBuf
   553→      // keep voices crisp: cut the sub-bass the engine plume occupies
   554→      const hp = ctx.createBiquadFilter()
   555→      hp.type = 'highpass'
   556→      hp.frequency.value = 190
   557→      const gain = ctx.createGain()
   558→      gain.gain.value = 0.0001
   559→      src.connect(hp)
   560→      hp.connect(gain)
   561→      gain.connect(this.master)
   562→      src.start(0)
   563→      this.chatReal = { src, gain }
   564→      gain.gain.setTargetAtTime(this.chatVol, ctx.currentTime, 0.5)
   565→      src.onended = () => {
   566→        if (this.chatReal && this.chatReal.src === src) this.chatReal = null
   567→        try { hp.disconnect() } catch { /* ignore */ }
   568→      }
   569→    } catch {
   570→      /* ignore */
   571→    }

   601→  playMachinery(volume = 0.12): void {
   602→    const v = clamp01(num(volume, 0.12))
   603→    const ctx = this.ctx
   604→    if (!ctx || !this.master || this.machBufs.length === 0 || v <= 0.001) return
   605→    if (this.shots >= this.SHOT_CAP) return
   606→    try {
   607→      const buf = this.machBufs[Math.floor(Math.random() * this.machBufs.length)]
   608→      const src = ctx.createBufferSource()
   609→      src.buffer = buf
   610→      src.playbackRate.value = 0.85 + Math.random() * 0.3
   611→      const lp = ctx.createBiquadFilter()
   612→      lp.type = 'lowpass'
   613→      lp.frequency.value = 2400 + Math.random() * 3200
   614→      const gain = ctx.createGain()
   615→      gain.gain.value = 0.0001
   616→      src.connect(lp)
   617→      lp.connect(gain)
   618→      const pan = ctx.createStereoPanner ? ctx.createStereoPanner() : null
   619→      let tail: AudioNode = gain
   620→      if (pan) {
   621→        pan.pan.value = Math.random() * 1.6 - 0.8
   622→        gain.connect(pan)
   623→        pan.connect(this.master)
   624→        tail = pan
   625→      } else {
   626→        gain.connect(this.master)
   627→      }
   628→      const dur = Math.min(buf.duration ?? 8, 6 + Math.random() * 9)
   629→      const off = Math.random() * Math.max(0.1, (buf.duration ?? 1) - dur - 0.2)
   630→      const t = ctx.currentTime + 0.03
   631→      gain.gain.setValueAtTime(0.0001, t)
   632→      gain.gain.setTargetAtTime(v, t, 1.1)
   633→      gain.gain.setTargetAtTime(0.0001, t + Math.max(0.5, dur - 1.6), 0.55)
   634→      src.start(t, off)
   635→      src.stop(t + dur + 0.4)
   636→      this.shots++
   637→      src.onended = () => {
   638→        this.shots--
   639→        try { tail.disconnect() } catch { /* ignore */ }
   640→      }
   641→    } catch {
   642→      /* ignore */
   643→    }

  1320→  private makeNoiseBuffer(kind: 'white' | 'brown'): AudioBuffer {
  1321→    const ctx = this.ctx
  1322→    if (!ctx) throw new Error('no context')
  1323→    const len = Math.floor(ctx.sampleRate * 2)
  1324→    const buf = ctx.createBuffer(1, len, ctx.sampleRate)
  1325→    const d = buf.getChannelData(0)
  1326→    if (kind === 'white') {
  1327→      for (let i = 0; i < len; i++) d[i] = Math.random() * 2 - 1
  1328→    } else {
  1329→      // brown noise: integrated white, normalized
  1330→      let last = 0
  1331→      let peak = 0
  1332→      for (let i = 0; i < len; i++) {
  1333→        const w = Math.random() * 2 - 1
  1334→        last = (last + 0.02 * w) / 1.02
  1335→        d[i] = last
  1336→        const a = Math.abs(last)
  1337→        if (a > peak) peak = a
  1338→      }
  1339→      if (peak > 1e-6) {
  1340→        const k = 0.9 / peak
  1341→        for (let i = 0; i < len; i++) d[i] *= k
  1342→      }
  1343→    }
  1344→    return buf
  1345→  }

  1447→  private applyAmbience(): void {
  1448→    const ctx = this.ctx
  1449→    if (!ctx || !this.ready) return
  1450→    const t = ctx.currentTime
  1451→    const { inside, planet, windIntensity, storm } = this.env
  1452→    const wi = windIntensity
  1453→
  1454→    // planet timbre (rare change)
  1455→    if (this.windFilter && this.windWhiteG && this.windBrownG && this.apPlanet !== planet) {
  1456→      this.apPlanet = planet
  1457→      try {
  1458→        if (planet === 'moon') {
  1459→          // thin / airy
  1460→          this.windFilter.type = 'bandpass'
  1461→          this.windFilter.frequency.setTargetAtTime(1050, t, 0.4)
  1462→          this.windFilter.Q.setTargetAtTime(0.55, t, 0.4)
  1463→          this.windWhiteG.gain.setTargetAtTime(0.9, t, 0.5)
  1464→          this.windBrownG.gain.setTargetAtTime(0.1, t, 0.5)
  1465→        } else {
  1466→          // lower / rumblier
  1467→          this.windFilter.type = 'lowpass'
  1468→          this.windFilter.frequency.setTargetAtTime(470, t, 0.4)
  1469→          this.windFilter.Q.setTargetAtTime(0.9, t, 0.4)
  1470→          this.windWhiteG.gain.setTargetAtTime(0.3, t, 0.5)
  1471→          this.windBrownG.gain.setTargetAtTime(0.65, t, 0.5)
  1472→        }
  1473→      } catch {
  1474→        /* ignore */
  1475→      }
  1476→    }
  1477→
  1478→    // wind level (muffled + reduced when inside; storms raise it a lot)
  1479→    let wind = inside ? 0.05 + wi * 0.03 : 0.05 + wi * 0.2
  1480→    if (planet === 'moon') wind *= 0.85
  1481→    if (storm) wind = inside ? Math.max(wind, 0.15) : Math.min(0.55, wind * 2.3 + 0.13)
  1482→    if (this.windGain && Math.abs(wind - this.apWind) > 0.003) {
  1483→      this.apWind = wind
  1484→      try {
  1485→        this.windGain.gain.setTargetAtTime(wind, t, inside ? 0.4 : 0.9)
  1486→      } catch {
  1487→        /* ignore */
  1488→      }
  1489→    }
  1490→
  1491→    // storm rattle layer
  1492→    const stormLvl = storm ? (inside ? 0.13 : 0.32 + wi * 0.2) : 0
  1493→    if (this.stormGain && Math.abs(stormLvl - this.apStorm) > 0.003) {
  1494→      this.apStorm = stormLvl
  1495→      try {
  1496→        this.stormGain.gain.setTargetAtTime(stormLvl, t, 1.1)
  1497→      } catch {
  1498→        /* ignore */
  1499→      }
  1500→    }
  1501→
  1502→    // room tone crossfade
  1503→    const room = inside ? 1 : 0
  1504→    if (this.roomGain && room !== this.apRoom) {
